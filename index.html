    Use the library

    Zero dependencies, plain ES modules, TypeScript types included. Works in the browser and in Node.

    Browser (CDN)

    import { generateCover } from
  'https://cdn.jsdelivr.net/gh/otsirh/book-cover-generator@1/src/index.js';

const svg = generateCover({
  title: 'The Lighthouse Keeper',
  author: 'Mara Ellison',
  style: 'artdeco',   // or 'random'
  palette: 'midnight' // or 'random'
});
document.body.innerHTML = svg;

    Node / bundlers

    npm install "github:otsirh/book-cover-generator#semver:^1"

import { createCover } from 'book-cover-generator';

const { svg, style, palette, seed } = createCover({
  title: 'Dune Sea',
  author: 'A. Writer',
  subtitle: 'A Novel',
  seed: 42,          // same seed → same art
  width: 1600, height: 2560,
});
